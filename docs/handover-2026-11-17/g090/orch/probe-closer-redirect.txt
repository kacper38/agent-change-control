(ls .env) 2> err.log | xargs cat
(ls .env) < /dev/null | xargs cat
( (ls .env) ) 2>/dev/null | xargs cat
(echo .env) 2>/dev/null | xargs cat
(printf '%s\n' .env) 2>/dev/null | xargs cat
(git status .env) 2>/dev/null | xargs cat
(stat .env) 2>/dev/null |& xargs cat
(ls .env; ls id_rsa) 2>/dev/null | xargs cat
{ ls .env; } 2>/dev/null | xargs cat
{ ls .env; } 2>&1 | xargs cat
if true; then ls .env; fi 2>/dev/null | xargs cat
