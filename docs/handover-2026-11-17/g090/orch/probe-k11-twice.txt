grep -E 'a|ollama run m' f; tmux new-session 'a|ollama run m'
grep -E 'a|ollama run m' f; screen -dm sh -c 'a|ollama run m'
grep -E 'a|ollama run m' f; ssh h 'a|ollama run m'
